// Re-export external API functions
export { getExternalData } from "./functions/getExternalData";
export { postExternalData } from "./functions/postExternalData";

// Re-export cron functions
export { dailyMorningJob } from "./functions/dailyMorningJob";
export { threeHourlyJob } from "./functions/threeHourlyJob";
import { dailyMorningJobDocs } from "./functions/dailyMorningJob";
import { threeHourlyJobDocs } from "./functions/threeHourlyJob";

// Import docs from each function
import { getExternalDataDocs } from "./functions/getExternalData";
import { postExternalDataDocs } from "./functions/postExternalData";
/**
 * Complete API documentation
 */
const apiDocs = {
  title: "Serverless Lambda API",
  description: "Complete documentation for the serverless lambda functions",
  version: "1.0.0",
  sections: {
    external: {
      title: "External API Integration",
      description: "Endpoints for interacting with external APIs",
      endpoints: {
        getExternalData: getExternalDataDocs,
        postExternalData: postExternalDataDocs,
      },
    },
    cron: {
      title: "Scheduled Jobs",
      description: "Automated scheduled tasks and cron jobs",
      endpoints: {
        dailyMorningJob: dailyMorningJobDocs,
        threeHourlyJob: threeHourlyJobDocs,
      },
    },
  },
};

/**
 * GET endpoint to serve API documentation
 * This Lambda function returns the complete API documentation in JSON format
 */
export const getDocs = async (event: any) => {
  try {
    return {
      statusCode: 200,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
        "Access-Control-Allow-Credentials": true,
      },
      body: JSON.stringify(apiDocs, null, 2),
    };
  } catch (error) {
    console.error("Error serving docs:", error);
    return {
      statusCode: 500,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
        "Access-Control-Allow-Credentials": true,
      },
      body: JSON.stringify({
        error: error instanceof Error ? error.message : "Unknown error",
      }),
    };
  }
};
