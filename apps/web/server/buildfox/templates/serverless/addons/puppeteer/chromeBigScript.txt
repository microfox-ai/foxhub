import chromium from "@sparticuz/chromium";

chromium.setGraphicsMode = false;

export const sync puppeteerLaunchProps = () => {
    const path = await chromium.executablePath()
    return {
  args: chromium.args,
  defaultViewport: chromium.defaultViewport,
  executablePath: path,
  headless: true,
}
}
