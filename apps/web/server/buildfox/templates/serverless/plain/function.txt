import dotenv from "dotenv";
import { APIGatewayProxyEvent, APIGatewayProxyResult } from "aws-lambda";

dotenv.config();

/**
 * POST endpoint to fetch data from an external API
 */
export const postExternalData = async (
  event: APIGatewayProxyEvent
): Promise<APIGatewayProxyResult> => {
  try {
    // Parse request body
    const body = event.body ? JSON.parse(event.body) : {};
    const { name, email, signupDate } = body;

    // Validate required fields
    if (!name || !email || !signupDate) {
      return {
        statusCode: 400,
        headers: {
          "Content-Type": "application/json",
          "Access-Control-Allow-Origin": "*",
        },
        body: JSON.stringify({
          error: "Missing required fields: name, email, and signupDate are required",
        }),
      };
    }

    // POST query parameters from the event
    const queryParams = event.queryStringParameters || {};
    const postId = queryParams.id ? parseInt(queryParams.id) : 1;

    // Fetch data from the API
    const response = await fetch(
      `https://jsonplaceholder.typicode.com/posts/${postId}`
    );
    const data = await response.json();

    return {
      statusCode: 200,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
      },
      body: JSON.stringify({
        data,
        userInfo: {
          name,
          email,
          signupDate,
        },
        timestamp: new Date().toISOString(),
      }),
    };
  } catch (error) {
    console.error("Error processing request:", error);
    return {
      statusCode: 500,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
      },
      body: JSON.stringify({
        error: error instanceof Error ? error.message : "Unknown error",
      }),
    };
  }
};
