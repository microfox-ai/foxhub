import dotenv from "dotenv";
import {
  getPaginationState,
  incrementPaginationState,
  getExternalDataRedisKey,
} from "./helpers/redis.js";

/**
 * Cron job that runs every day at 8:00 AM
 */
export const dailyMorningJob = async (event: any) => {
  try {
    // Clean up old records (30 days)
    const paginationState = await getPaginationState(getExternalDataRedisKey);

    await incrementPaginationState(getExternalDataRedisKey);
    return {
      statusCode: 200,
      body: JSON.stringify({
        message:
          "Daily morning job completed successfully - incrementing the page by 1",
        currentPage: paginationState.page,
      }),
    };
  } catch (error) {
    console.error("Error running daily job:", error);
    return {
      statusCode: 500,
      body: JSON.stringify({
        error: error instanceof Error ? error.message : "Unknown error",
      }),
    };
  } finally {
  }
};

 * Documentation for cron job endpoints
 */
