import dotenv from "dotenv";
import { APIGatewayProxyResult } from "aws-lambda";
import {
  getPaginationState,
  incrementPaginationState,
  getExternalDataRedisKey,
} from "./helpers/redis.js";

dotenv.config();

/**
 * Cron job that runs every day at 8:00 AM to manage pagination state
 * This job ensures the pagination state is properly maintained and cleaned up
 */
export const dailyMorningJob = async (): Promise<APIGatewayProxyResult> => {
  try {
    // Get current pagination state
    const paginationState = await getPaginationState(getExternalDataRedisKey);

    // Increment the page counter
    await incrementPaginationState(getExternalDataRedisKey);

    return {
      statusCode: 200,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
      },
      body: JSON.stringify({
        message: "Daily morning job completed successfully",
        details: {
          previousPage: paginationState.page,
          newPage: paginationState.page + 1,
          lastUpdated: new Date().toISOString(),
        },
      }),
    };
  } catch (error) {
    console.error("Error running daily job:", error);
    return {
      statusCode: 500,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
      },
      body: JSON.stringify({
        error: error instanceof Error ? error.message : "Unknown error",
        timestamp: new Date().toISOString(),
      }),
    };
  }
};
