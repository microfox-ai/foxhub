import { Redis } from "@upstash/redis";

// Initialize Redis client
const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!,
});

/**
 * Cron job that runs every day at 8:00 AM
 */
export const dailyMorningJob = async (event: any) => {
  try {
    // Clean up old records (30 days)
    const date = new Date();
    date.setDate(date.getDate() - 30);

    // Example: Get all keys older than 30 days and delete them
    const keys = await redis.keys("*");
    let deletedCount = 0;

    for (const key of keys) {
      // Example: Get the timestamp of the key
      const timestamp = await redis.get(`${key}:timestamp`);
      if (timestamp && new Date(timestamp) < date) {
        // Example: Delete the key and the timestamp
        await redis.del(key);
        await redis.del(`${key}:timestamp`);
        deletedCount++;
      }
    }

    return {
      statusCode: 200,
      body: JSON.stringify({
        message: "Daily morning job completed successfully",
        deletedRecords: deletedCount,
      }),
    };
  } catch (error) {
    console.error("Error running daily job:", error);
    return {
      statusCode: 500,
      body: JSON.stringify({
        error: error instanceof Error ? error.message : "Unknown error",
      }),
    };
  } finally {
  }
};
