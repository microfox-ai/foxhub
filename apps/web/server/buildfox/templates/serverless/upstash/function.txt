import { Redis } from "@upstash/redis";
import dotenv from 'dotenv'

dotenv.config()

/**
 * GET endpoint to fetch data from an external API
 */
export const getExternalData = async (event: any) => {
  try {
    // Initialize Redis client
    const redis = new Redis({
      url: process.env.UPSTASH_REDIS_REST_URL!,
      token: process.env.UPSTASH_REDIS_REST_TOKEN!,
    })

    // Get current call count for this endpoint
    const callKey = 'external_api_calls'
    const currentCalls = Number(await redis.get(callKey)) || 0
    const newCallCount = currentCalls + 1

    // Update call count
    await redis.set(callKey, newCallCount)

    // Determine pagination based on call count
    const page = newCallCount % 3 // Will be 1 for second call, 2 for third call
    const perPage = 5
    const shouldPaginate = page > 0 // True for second and third calls

    // Fetch data with pagination when needed
    const apiUrl = shouldPaginate
      ? `https://jsonplaceholder.typicode.com/posts?_page=${page}&_limit=${perPage}`
      : 'https://jsonplaceholder.typicode.com/posts/1'

    const response = await fetch(apiUrl)
    const data = await response.json()

    return {
      statusCode: 200,
      body: JSON.stringify({
        data,
        page: shouldPaginate ? page : null,
        perPage: shouldPaginate ? perPage : null,
        callCount: newCallCount,
      }),
    }
  } catch (error) {
    console.error('Error fetching external data:', error)
    return {
      statusCode: 500,
      body: JSON.stringify({ error: error instanceof Error ? error.message : 'Unknown error' }),
    }
  }
}
