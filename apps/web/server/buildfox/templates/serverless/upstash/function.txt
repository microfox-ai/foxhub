import dotenv from "dotenv";
import { APIGatewayProxyEvent, APIGatewayProxyResult } from "aws-lambda";
import {
  getPaginationState,
  incrementPaginationState,
  getExternalDataRedisKey,
} from "./helpers/redis.js";

dotenv.config();

/**
 * POST endpoint to fetch data from an external API
 */
export const postExternalData = async (
  event: APIGatewayProxyEvent
): Promise<APIGatewayProxyResult> => {
  try {
    // Parse request body
    const body = event.body ? JSON.parse(event.body) : {};
    const { name, email, signupDate } = body;

    // Validate required fields
    if (!name || !email || !signupDate) {
      return {
        statusCode: 400,
        headers: {
          "Content-Type": "application/json",
          "Access-Control-Allow-Origin": "*",
        },
        body: JSON.stringify({
          error: "Missing required fields: name, email, and signupDate are required",
        }),
      };
    }

    // Get query parameters from the event
    const queryParams = event.queryStringParameters || {};
    const requestedPage = queryParams.page
      ? parseInt(queryParams.page)
      : undefined;
    const requestedPerPage = queryParams.perPage
      ? parseInt(queryParams.perPage)
      : undefined;

    // Initialize Redis client
    const paginationState = await getPaginationState(getExternalDataRedisKey);

    // Use provided parameters or fall back to Redis state
    const page = requestedPage || paginationState.page;
    const perPage = requestedPerPage || paginationState.limit;
    const shouldPaginate = page > 0;

    // Fetch data with pagination when needed
    const apiUrl = shouldPaginate
      ? `https://jsonplaceholder.typicode.com/posts?_page=${page}&_limit=${perPage}`
      : "https://jsonplaceholder.typicode.com/posts/1";

    const response = await fetch(apiUrl);
    const data = await response.json();

    // Only increment Redis state if we're using the default pagination
    if (!requestedPage && !requestedPerPage) {
      await incrementPaginationState(getExternalDataRedisKey);
    }

    return {
      statusCode: 200,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
      },
      body: JSON.stringify({
        data,
        page: shouldPaginate ? page : null,
        perPage: shouldPaginate ? perPage : null,
        callCount: paginationState.page,
      }),
    };
  } catch (error) {
    console.error("Error fetching external data:", error);
    return {
      statusCode: 500,
      headers: {
        "Content-Type": "application/json",
        "Access-Control-Allow-Origin": "*",
      },
      body: JSON.stringify({
        error: error instanceof Error ? error.message : "Unknown error",
      }),
    };
  }
};
