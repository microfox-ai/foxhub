import dotenv from "dotenv";
import {
  getPaginationState,
  incrementPaginationState,
  getExternalDataRedisKey,
} from "./helpers/redis.js";

dotenv.config();

/**
 * GET endpoint to fetch data from an external API
 */
export const getExternalData = async (event: any) => {
  try {
    // Initialize Redis client
    const paginationState = await getPaginationState(getExternalDataRedisKey);

    // Determine pagination based on call count
    const page = paginationState.page;
    const perPage = paginationState.limit;
    const shouldPaginate = page > 0;

    // Fetch data with pagination when needed
    const apiUrl = shouldPaginate
      ? `https://jsonplaceholder.typicode.com/posts?_page=${page}&_limit=${perPage}`
      : "https://jsonplaceholder.typicode.com/posts/1";

    const response = await fetch(apiUrl);
    const data = await response.json();

    await incrementPaginationState(getExternalDataRedisKey);

    return {
      statusCode: 200,
      body: JSON.stringify({
        data,
        page: shouldPaginate ? page : null,
        perPage: shouldPaginate ? perPage : null,
        callCount: paginationState.page,
      }),
    };
  } catch (error) {
    console.error("Error fetching external data:", error);
    return {
      statusCode: 500,
      body: JSON.stringify({
        error: error instanceof Error ? error.message : "Unknown error",
      }),
    };
  }
};
